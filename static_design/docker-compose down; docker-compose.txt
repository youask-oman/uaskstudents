$ts = Get-Date -Format "yyyyMMdd_HHmmss"
$cid = docker compose ps -q postgres
$remote = "/tmp/uask_db_$ts.dump"
$local  = ".\uask_db_$ts.dump"

docker compose exec -T postgres pg_dump -U uask_user -Fc -d uask_db -f $remote
docker cp "${cid}:${remote}" $local
docker compose exec -T postgres pg_restore -l $remote > $null

Write-Host "OK: $local"
